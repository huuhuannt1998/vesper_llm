% ============================================================================
% KEY EQUATIONS
% ============================================================================

% Proximity-based interaction detection
\begin{equation}
    \text{Interact}(o, a) = \begin{cases}
        \text{True}, & \text{if } \|p_a - p_o\| \leq d_o \\
        \text{False}, & \text{otherwise}
    \end{cases}
\end{equation}

% Time acceleration scale calculation
\begin{equation}
    s = \frac{t_v}{t_r^{\max}}
\end{equation}

% Virtual timestamp calculation
\begin{equation}
    t_v(t_r) = t_0 + s \cdot (t_r - t_{r0})
\end{equation}
